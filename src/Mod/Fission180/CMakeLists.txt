# Fission 180 module: the Fusion-style interface layer and the golden-model suite.
# Pure Python for now, so this only copies files into the build and install trees.

# VERSION: Fission 180's own version ("0.1.0-beta"), the one file the programs (src/Main), the
# installer, the release scripts and fission180.__version__ all read.
SET(Fission180_Init
    Init.py
    InitGui.py
    VERSION
)

# Every module, icon and test file in the folder (never a list kept by hand: in October
# 2026 the list still had 35 of 116 modules, the full build missed the other 81 and
# failed with "cannot import name 'blend' from 'fission180'"; the quick update copies the
# whole folder, so it never showed there). tools/fission180/check_install.sh checks this
# on every push. CONFIGURE_DEPENDS: a new file is picked up without re-running CMake by
# hand. Recursive: sub-packages such as fission180/ai/ops/ are included.
FILE(GLOB_RECURSE Fission180_Package RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    fission180/*.py)
FILE(GLOB Fission180_Icons RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    fission180/icons/*.svg)

# Ribbon/menu icons, generated by tools/fission180/make_icons.py.
FILE(GLOB Fission180_UiIcons RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    fission180/icons/ui/*.svg)

# Unit tests, golden models (their builder and the model files) and the GUI tests.
FILE(GLOB_RECURSE Fission180_Tests RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    Fission180Tests/*.py Fission180Tests/*.json Fission180Tests/*.md Fission180Tests/*.txt)

# Sample files the Insert/Export tests open (meshes, STEP, SVG, DXF, images): every file.
# (files only: the VEX test parts' generator sits in a sub-folder, installed with the tests)
FILE(GLOB Fission180_Fixtures LIST_DIRECTORIES false RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
    CONFIGURE_DEPENDS Fission180Tests/fixtures/*)

# What the Home tab's welcome line opens offline: the cheat sheet (a copy of
# docs/fission180/cheat-sheet.md; a unit test keeps the two the same) and the Raspberry Pi 5
# case design (a copy of docs/fission180/examples/pi5-case/pi5_case.FCStd).
# docs/ai/ (recursive): the documents for AI models the MCP server offers as resources
# (fission180/ai/mcp_resources.py).
FILE(GLOB_RECURSE Fission180_Docs RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS docs/*.md)
FILE(GLOB_RECURSE Fission180_Examples RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    examples/*.FCStd)

SET(Fission180_ALL ${Fission180_Init} ${Fission180_Package} ${Fission180_Icons} ${Fission180_UiIcons}
    ${Fission180_Docs} ${Fission180_Examples} ${Fission180_Tests} ${Fission180_Fixtures})

ADD_CUSTOM_TARGET(Fission180 ALL SOURCES ${Fission180_ALL})

fc_copy_sources(Fission180 "${CMAKE_BINARY_DIR}/Mod/Fission180" ${Fission180_ALL})

INSTALL(FILES ${Fission180_Init} DESTINATION Mod/Fission180)
# DIRECTORY keeps the sub-folders (INSTALL(FILES) would put fission180/ai/ops/core.py into
# fission180/). The icons have their own rules below.
INSTALL(DIRECTORY fission180 DESTINATION Mod/Fission180
    FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
INSTALL(FILES ${Fission180_Icons} DESTINATION Mod/Fission180/fission180/icons)
INSTALL(FILES ${Fission180_UiIcons} DESTINATION Mod/Fission180/fission180/icons/ui)
INSTALL(DIRECTORY Fission180Tests DESTINATION Mod/Fission180
    FILES_MATCHING PATTERN "*.py" PATTERN "*.json" PATTERN "*.md" PATTERN "*.txt"
    PATTERN "__pycache__" EXCLUDE)
INSTALL(FILES ${Fission180_Fixtures} DESTINATION Mod/Fission180/Fission180Tests/fixtures)
INSTALL(DIRECTORY docs examples DESTINATION Mod/Fission180
    FILES_MATCHING PATTERN "*.md" PATTERN "*.FCStd")
